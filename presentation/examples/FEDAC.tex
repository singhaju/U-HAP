% 3. PROBLEM STATEMENT
% ════════════════════════════════════════════════════════════════════════════
\begin{frame}{Problem Statement}
\vspace{2pt}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\begin{callout}{Hot-path latency}
Per-request policy fetch \& eval adds \textbf{tens of ms} tail latency to every microservice call.
\end{callout}\vspace{4pt}
\begin{callout}{Unverifiable policy governance}
No tamper-evident audit mechanism to detect or enforce policy changes across domains.
\end{callout}
\end{column}
\begin{column}{0.49\textwidth}
\begin{callout}{Inefficient authn / authz pipeline}
No unified authn/authz pipeline scales across heterogeneous, multi-domain K8s clusters.
\end{callout}\vspace{4pt}
\begin{callout}{Lack of privacy-preserving identity}
Persistent identity exposure across services enables cross-domain tracking \& profiling.
\end{callout}
\end{column}
\end{columns}
\end{frame}

% ════════════════════════════════════════════════════════════════════════════
% 4. OUR SOLUTION — FEDAC
% ════════════════════════════════════════════════════════════════════════════
\begin{frame}{Our Solution: FEDAC}
\vspace{0pt}
\begin{columns}[T,onlytextwidth]
\begin{column}{\textwidth}
\footnotesize
\begin{itemize}
  \setlength{\topsep}{1pt}\setlength{\itemsep}{6pt}\setlength{\parsep}{0pt}
  \item \textbf{Hub-and-spoke IAM} — Hub reasons policy centrally%; sidecars enforce O(1) locally with no per-request policy fetch.
  \item \textbf{Risk-adaptive auth (T1–T4)} — selected auth type per service sensitivity%T1 SSO $\to$ T2 MFA $\to$ T3 context-risk $\to$ T4 ZKP, selected per service sensitivity; partial failure denies only failing services.
  \item \textbf{Privacy-preserving identity} — Session-scoped pseudonymous UIDs bound per OID set%; cross-domain correlation impossible.
  \item \textbf{Authorization-as-Result (AaR)} — Auth decision baked into a short-lived, service-scoped token%; O(1) in-process sidecar verify.
  \item \textbf{Merkle-anchored governance} — Policies on IPFS%; integrity anchored on-chain. Incremental $O(\log N)$ updates.
  \item \textbf{Targeted revocation} — Only tokens whose OID set intersects the changed policy are invalidated%; all others stay valid.
\end{itemize}
\end{column}

% 7. LAYER 1 — IDENTITY FEDERATION (P1–P3)
% ════════════════════════════════════════════════════════════════════════════
\begin{frame}{Layer 1: Identity Federation Layer}
\begin{columns}[c,onlytextwidth] % <-- change T → c

\begin{column}{0.55\textwidth}
\vfill
\centering
\includegraphics[width=8cm]{figs/layer1.png}
\vfill
\end{column}

\begin{column}{0.42\textwidth}
\begin{itemize}\tightitems
\item \textbf{P1:} OID-classified multi-resource request received at the Hub; security tier derived per service.
\item \textbf{P2:} Dynamic Authentication Manager selects tier per OID: T1 SSO $\to$ T2 MFA $\to$ T3 context-risk $\to$ T4 ZKP. Each tier executes independently; partial auth denies only failing services.
\item \textbf{P3:} Identity normalization — context-bound pseudonymous UID derived from session nonce; prevents cross-domain correlation.
\end{itemize}
\end{column}

\end{columns}
\end{frame}

% 8. LAYER 2 — POLICY GOVERNANCE (P4)
% ════════════════════════════════════════════════════════════════════════════
\begin{frame}{Layer 2: Policy Governance Layer}
\vspace{-0.2cm}
\begin{columns}[c,onlytextwidth] % T → c

\begin{column}{0.55\textwidth}
\vfill
\centering
\includegraphics[width=8cm]{figs/layer2.png}
\vfill
\end{column}

\begin{column}{0.42\textwidth}
\scriptsize
\begin{itemize}\tightitems
\item \textbf{P4:} Policy Engine fetches policies from IPFS via content-addressed CID.
\item Merkle Tree hashes leaves; Validation Engine checks recomputed root against the on-chain anchor.
\item No policy consumed without a valid Merkle proof — tamper-evident by construction.
\item Three smart contracts: root registry, UID-to-policy map, token audit log.
\end{itemize}
\end{column}

\end{columns}
\end{frame}

\begin{frame}{Layer 3: Authorization and Enforcement Layer}
\vspace{-0.3cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.55\textwidth}
\centering{\includegraphics[width=8cm]{figs/layer3.png}}
\end{column}

\begin{column}{0.42\textwidth}
\scriptsize
\begin{itemize}\tightitems
\item \textbf{P5:} AaR token — Ed25519 signed, AES-256-GCM encrypted, scoped to OID + allowed ops.
\item \textbf{P6:} Sidecar verifies in-process — O(1). No IPFS, no chain, no policy fetch on hot path.
\item \textbf{P7:} Token Extender renews from baked conditions; no IPFS or blockchain (Inv.~2).
\item \textbf{P8:} OID-intersection revocation on policy update; unaffected tokens stay valid. $O(\log N)$ Merkle (Inv.~3).
\end{itemize}
\end{column}
\end{columns}
\end{frame}

% 21. CONCLUSION

\begin{frame}{Conclusion}
\vspace{2pt}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\begin{callout}{Separate expensive from cheap}
\tiny
Authorization reasoning is costly — policy retrieval, Merkle verification, identity normalization. FEDAC does this \emph{once}, at the Hub, at issuance. Every subsequent request pays only for a local cryptographic check — no network, no re-evaluation.
\end{callout}\vspace{2pt}
\begin{callout}{Make the decision portable}
\tiny
The AaR token carries the authorization decision itself — not a reference to it. A sidecar that loses connectivity still enforces correctly. Renewal re-reads conditions baked into the token, not from IPFS.
\end{callout}
\end{column}
\begin{column}{0.49\textwidth}
\begin{callout}{Keep governance tamper-evident}
\tiny
Every policy is a Merkle leaf anchored on-chain. Any modification — accidental or adversarial — is detected before the policy is consumed. Revocation targets only the OID intersection: no legitimate token is ever invalidated by a policy change it is not subject to.
\end{callout}\vspace{2pt}
\begin{callout}{Scope everything to the service}
\tiny
Identity, capability, and revocation are all scoped to the specific service being accessed. The same user has cryptographically unlinkable identities across services. A compromised service cannot escalate to another.
\end{callout}
\end{column}
\end{columns}
% \vspace{3pt}
% \pinkbanner{Fine-grained authorization should not be a performance tax. FEDAC proves it does not have to be.}
\end{frame}

\begin{frame}[noframenumbering]{Backup: Phase 6 Sidecar Verify (Edge Enforcement)}
\vspace{4pt}
\begin{callout}{Algorithm sketch}
\small
\begin{plainsteps}
\item \textbf{1.} Parse incoming \texttt{Authorization: Bearer} AaR token.
\item \textbf{2.} Verify Ed25519 signature using the Hub public key.
\item \textbf{3.} Decrypt private payload with AES-256-GCM.
\item \textbf{4.} Check expiry and audience OID against the receiving service.
\item \textbf{5.} Check requested operation against baked allowed-ops.
\item \textbf{6.} Forward on success; return HTTP 403 on failure.
\end{plainsteps}
\end{callout}
\vspace{4pt}
\pinkbanner{All checks are in-process; no IPFS, no blockchain, no policy server.}

\begin{callout}{When revocation triggers}

\begin{itemize}\tightitems
\item \textbf{Authorized update:} admin uploads $\Phi'_j$ to IPFS; Merkle branch recomputed; blockchain root updated.
\item \textbf{Integrity violation:} Hub finds $R' \neq R$; Active Revocation Protocol fires immediately.
\end{itemize}
\end{callout}
\begin{callout}{3 simultaneous actions (integrity violation)}
\tiny
\setlength{\tabcolsep}{2pt}
\renewcommand{\arraystretch}{1.05}
\begin{tabular}{>{\bfseries}l p{4.5cm}}
Alert     & Signed report of affected $PID_j/CID_j$ to admin \\
Quarantine & Issuance \& renewal suspended for $\mathbf{OID}_{comp}$ \\
Kill signal & $\mathcal{S}_{hub}(UID \| \mathbf{OID}_{comp})$ broadcast to all sidecars \\
\end{tabular}
\end{callout}
\end{column}
\begin{column}{0.44\textwidth}
\begin{callout}{OID-intersection targeting}
\tiny
\[
T_{active} \leftarrow T_{active} \setminus
\bigl\{\,t \mid t.\mathbf{OID} \cap \mathbf{OID}_{comp} \neq \emptyset\,\bigr\}
\]
\vspace{-4pt}
Only tokens overlapping the compromised set are revoked; all others stay valid.
\end{callout}

\end{callout}

\begin{frame}[noframenumbering]{Backup: Why Not Plain JWT / OIDC?}
\vspace{2pt}
\centering
\scriptsize
\setlength{\tabcolsep}{4pt}
\renewcommand{\arraystretch}{1.0}
\begin{tabular}{p{2.8cm} p{3.2cm} p{4.8cm}}
\toprule
\rowcolor{fedacGray}\textbf{Property} & \textbf{JWT (OIDC)} & \textbf{AaR (FEDAC)}\\
\midrule
Payload visibility    & All claims plaintext & AES-256-GCM encrypted payload; UID is pseudonymous \\
Service binding       & Audience is an app, not a service set & $aud = \mathbf{OID}$ — exact microservice set; cross-service replay blocked \\
Authorization scope   & Generic identity claims; app re-evaluates policy & Per-service $\mathbf{Cap}$ baked at issuance; sidecar reads, not re-evaluates \\
Hot-path verify       & JWKS RTT per key rotation & Ed25519 + SHA-256 in-process; no network call; O(1) \\
Renewal               & Full re-authentication (new OIDC flow) & Challenge-response from baked conditions; 2.34\,ms p50 \\
Revocation            & Blocklist or short TTL & Targeted OID-intersection kill-signal; unaffected tokens valid \\
Risk-adaptive auth    & Not defined & T1--T4 per service sensitivity; ZKP for high-sensitivity OIDs \\
\bottomrule
\end{tabular}\\[4pt]
\pinkbanner{AaR is not a JWT replacement — it is the authorization \emph{outcome}. The OIDC identity token remains the input to FEDAC Phase 1.}
